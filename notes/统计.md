// 配置
const FORCES = ["汉", "魏", "蜀", "吴", "群", "晋"];
const TYPES = ["君主", "野心家", "叛首", "叛谍", "客将", "隐士", "可君主升变"];

// 缓存所有页面
const pages = dv.pages().where(p => p.tags && p.tags.includes("武将"));
const pagesWithTags = pages.filter(p => p.tags && p.tags.length > 0);

// 统计工具类
class StatsCollector {
  constructor() {
    this.stats = new Map();
    this.initializeStats();
  }

  initializeStats() {
    // 初始化势力统计
    FORCES.forEach(f => {
      this.stats.set(`势力：${f}`, 0);
      this.stats.set(`客将：${f}`, 0);
      this.stats.set(`伪装：${f}`, 0);
    });
  }

  // 统计单个标签
  countTag(tag) {
    return pagesWithTags.filter(p => p.tags.includes(tag)).length;
  }

  // 统计同时具有两个标签的页面
  countMatching(baseTag, targetTag) {
    return pagesWithTags.filter(p =>
      p.tags.includes(baseTag) && p.tags.includes(targetTag)
    ).length;
  }

  // 生成交叉统计表格
  generateCrossTable(basePrefix, targetPrefix) {
    const header = [" "].concat(FORCES.map(f => `${targetPrefix}：${f}`));
    const rows = FORCES.map(base => {
      const row = [`${basePrefix}：${base}`];
      FORCES.forEach(target => {
        const baseTag = `${basePrefix}：${base}`;
        const targetTag = `${targetPrefix}：${target}`;
        row.push(this.countMatching(baseTag, targetTag));
      });
      return row;
    });
    return [header, ...rows];
  }

  // 获取势力统计表格数据
  getForceStats() {
    return FORCES.map(f => [
      f,
      this.countTag(`势力：${f}`),
      this.countTag(`客将：${f}`),
      this.countTag(`伪装：${f}`)
    ]);
  }

  // 获取武将类型统计
  getTypeStats() {
    return TYPES.map(type => [
      type,
      this.countTag(type)
    ]);
  }
}

// 执行统计
const collector = new StatsCollector();

// === 全势力角色数 ===
dv.header(3, "全势力武将数");
dv.table(
  ["势力", "主势力武将数", "客势力武将数", "伪势力叛谍数"],
  collector.getForceStats()
);

// === 各势力客将数 ===
dv.header(3, "各势力客将数");
// 自定义行首和列首
const customRowHeaders = ["汉", "魏", "蜀", "吴", "群", "晋"];
const customColHeaders = ["客汉", "客魏", "客蜀", "客吴", "客群", "客晋"];

function generateCustomTable(basePrefix, targetPrefix, rowHeaders, colHeaders, topLeft = " ") {
  // 使用自定义的左上角和列首
  const header = [topLeft].concat(colHeaders);
  const rows = FORCES.map((base, index) => {
    // 使用自定义的行首
    const row = [rowHeaders[index]];
    FORCES.forEach(target => {
      const baseTag = `${basePrefix}：${base}`;
      const targetTag = `${targetPrefix}：${target}`;
      row.push(collector.countMatching(baseTag, targetTag));
    });
    return row;
  });
  return [header, ...rows];
}

// 使用自定义表头，包括左上角
const guestTable = generateCustomTable("势力", "客将", customRowHeaders, customColHeaders, "主势力");
dv.table(guestTable[0], guestTable.slice(1));

// === 武将类型统计 ===
dv.header(3, "各武将类型");
dv.table(
  ["类型", "数量"],
  collector.getTypeStats()
);

// === 体力值统计 ===

// 统计容器
let totalStats = { 1.5: 0, 2: 0, 2.5: 0 };
let forceStats = {};
for (let f of FORCES) {
  forceStats[f] = { 1.5: 0, 2: 0, 2.5: 0 };
}

// 遍历所有武将
for (let p of pages) {
  let hp = p["体力值"];
  if ([1.5, 2, 2.5].includes(hp)) {
    // ✅ 先累计总数
    totalStats[hp]++;

    // ✅ 找到该武将的势力（tags 里有“势力：X”）
    let factions = (p.tags || []).filter(t => t.startsWith("势力："));
    for (let facTag of factions) {
      let f = facTag.replace("势力：", ""); // 取出 汉 / 魏 / ...
      if (FORCES.includes(f)) {
        forceStats[f][hp]++;
      }
    }
  }
}

// === 组合表格数据 ===
let header = ["势力", "1.5", "2", "2.5"];
let rows = FORCES.map(f => [
  f,
  forceStats[f][1.5],
  forceStats[f][2],
  forceStats[f][2.5]
]);

// 全武将体力分布
rows.push([
  "总计",
  totalStats[1.5],
  totalStats[2],
  totalStats[2.5]
]);

dv.header(3, "体力分布");
dv.table(header, rows);
